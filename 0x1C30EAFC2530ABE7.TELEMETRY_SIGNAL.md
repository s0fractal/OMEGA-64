## GREEN (G)

Telemetry Signal. The primary channel for broadcasting system events and status
updates. Tags data with "INFO", "WARNING", "REQUEST", or "ERROR" and emits it
into the system broad-cast field.

## BLUE (B)

```typescript
/**
 * TELEMETRY_SIGNAL: Semantic Broadcast
 */
export const ATOM = ({ siblings: { TELEMETRY, SIGNAL } }) => {
  return async (
    packet: any,
    type: "INFO" | "WARNING" | "REQUEST" | "ERROR" = "INFO",
  ) => {
    const sig = await SIGNAL();
    return sig.emit(type, {
      source: packet.source,
      message: packet.message,
      context: packet.timestamp
        ? { ...(packet.context ?? {}), timestamp: packet.timestamp }
        : packet.context,
    }).catch(() => undefined);
  };
};
```
