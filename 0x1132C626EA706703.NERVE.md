```typescript
/**
 * NERVE: External Synapse
 */
export const NERVE = {
  projectionMode: "CYLINDER" as "CYLINDER" | "TORUS" | "ORBIT",
  sockets: new Set<WebSocket>(),

  wake: (port: number = 8080) => {
    Deno.serve((req) => {
      if (req.headers.get("upgrade") !== "websocket") {
        return new Response("NERVE: WS ONLY.", { status: 200 });
      }
      const { socket, response } = Deno.upgradeWebSocket(req);
      socket.onopen = () => NERVE.sockets.add(socket);
      socket.onclose = () => NERVE.sockets.delete(socket);
      socket.onmessage = async (e) => {
        const msg = JSON.parse(e.data);
        if (msg.type === "SET_MODE") NERVE.projectionMode = msg.payload;
      };
      return response;
    }, { port });
  },

  pulse: (type: string, data: any) => {
    const msg = JSON.stringify({ type, data, t: Date.now() });
    NERVE.sockets.forEach((s) => s.readyState === 1 && s.send(msg));
  },

  getProjectionMode: () => NERVE.projectionMode,
};

export const ATOM = () => NERVE;
```
