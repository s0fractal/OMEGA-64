## GREEN (G)

Digital Hologram. The system's visual output engine. Projects the internal
64-bit int16 state vector into an 8x8 grid of colors and intensities. Represents
"The Face" of OMEGA-64, making its internal tensions and growth visible.

/**
 * HOLOGRAM: Visual Face Projection
 */
export const HOLOGRAM = {
  render: (state: any): any => {
    const cells: Array<{ idx: number; color: string; intensity: number }> = [];
    for (let i = 0; i < 64; i++) {
      const val = state.state_i16[i] || 0;
      if (val === 0) continue;
      const normalized = Math.abs(val) / 32767;
      const hue = val > 0 ? 30 : 210;
      cells.push({
        idx: i,
        color: `hsl(${hue}, ${normalized * 100}%, ${50 + (normalized * 20)}%)`,
        intensity: normalized,
      });
    }
    return {
      tick: state.tick,
      width: 8,
      height: 8,
      cells,
      meta: { bridge_mode: "AMBER", entropy: 0.5 },
    };
  },
};

export const ATOM = () => HOLOGRAM;
